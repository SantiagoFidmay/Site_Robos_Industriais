desafio_01_iluminacao.ino 
// DESAFIO 01 — ILUMINAÇÃO INTELIGENTE — PIR + LDR + LED
const int pinoLED  = 13;
const int pinoPIR  = 7;
const int pinoLDR  = A0;
const int limiarLDR = 500;

void setup() {
  pinMode(pinoLED, OUTPUT);
  pinMode(pinoPIR, INPUT);
  Serial.begin(9600);
}

void loop() {
  int estadoPIR = digitalRead(pinoPIR);
  int valorLDR  = analogRead(pinoLDR);

  if (estadoPIR == HIGH && valorLDR < limiarLDR) {
    digitalWrite(pinoLED, HIGH);
    Serial.println("[ON] Movimento + Escuro → LED LIGADO");
  } else {
    digitalWrite(pinoLED, LOW);
    Serial.println("[OFF] Condição não atendida");
  }
  delay(200);
}

// DESAFIO 02 — DISTÂNCIA E ILUMINAÇÃO — Ultrassônico + POT + LED
const int pinoTrig = 9;
const int pinoEcho = 8;
const int pinoLED  = 13;
const int pinoPOT  = A0;

float medirDistancia() {
  digitalWrite(pinoTrig, LOW);  delayMicroseconds(2);
  digitalWrite(pinoTrig, HIGH); delayMicroseconds(10);
  digitalWrite(pinoTrig, LOW);
  long d = pulseIn(pinoEcho, HIGH);
  return (d * 0.034) / 2.0;
}

void loop() {
  float dist  = medirDistancia();
  float limite = map(analogRead(pinoPOT), 0, 1023, 10, 100);
  digitalWrite(pinoLED, dist < limite ? HIGH : LOW);
  delay(300);
}

desafio_03_ambiente.ino 
// DESAFIO 03 — AMBIENTE INTELIGENTE — DHT + LEDs indicadores
#include <DHT.h>
#define PINO_DHT  4
#define TIPO_DHT  DHT11
DHT dht(PINO_DHT, TIPO_DHT);

const int ledVerde    = 10;
const int ledVermelho = 11;
const int ledAmarelo  = 12;
const float TEMP_MAX = 30.0;
const float UMID_MIN = 40.0;

void loop() {
  float t = dht.readTemperature();
  float u = dht.readHumidity();
  digitalWrite(ledVerde,    (!(t > TEMP_MAX) && !(u < UMID_MIN)) ? HIGH : LOW);
  digitalWrite(ledVermelho, (t > TEMP_MAX) ? HIGH : LOW);
  digitalWrite(ledAmarelo,  (u < UMID_MIN) ? HIGH : LOW);
  delay(2000);
}

FUNDAMENTOS

CONCEITOS — IoT & ROBÓTICA INDUSTRIAL

Compreenda os pilares que sustentam a revolução industrial moderna e como robótica e IoT se integram na Indústria 4.0.

Internet das Coisas (IoT)

A IoT refere-se à interconexão digital de objetos físicos com a internet por meio de sensores, atuadores e softwares embarcados. No contexto industrial (IIoT), máquinas e robôs coletam e compartilham dados em tempo real via protocolos como MQTT, OPC UA e AMQP.

Robótica Industrial

Ramo da engenharia que envolve o projeto e aplicação de robôs em ambientes de manufatura. Segundo a ISO 8373:2012, um robô industrial é um manipulador multipropósito, reprogramável em três ou mais eixos, que executa tarefas como soldagem, montagem e inspeção com alta precisão 24h/dia.

Importância da Automação

A automação industrial proporciona: aumento de produtividade (24/7), melhoria de qualidade, redução de custos, maior segurança dos trabalhadores, flexibilidade de produção e rastreabilidade completa. Na Indústria 4.0, combina robótica, IA e IoT para fábricas autônomas.

Robótica & IoT: Convergência

A convergência cria sistemas ciberfísicos onde robôs são nós inteligentes em rede. Sensores IoT coletam dados de vibração, temperatura e performance enviando para a nuvem. Isso permite Digital Twins, manutenção preditiva por machine learning e controle remoto via nuvem.

Arduino e Prototipagem IoT

Plataforma open-source baseada em microcontroladores que leem inputs de sensores e executam lógica programada em C/C++. É vastamente utilizada para validar projetos IoT e de automação, suportando dezenas de bibliotecas para quase todos os sensores do mercado.

Sensores na Automação

Sensores são os "sentidos" da fábrica inteligente. O fluxo da automação começa no sensor (aquisição), viaja através de protocolos de comunicação (I2C, SPI, UART, MQTT) e chega ao controlador, que processa e atua no sistema — pilar da IoT que conecta o mundo físico ao digital.

EVOLUÇÃO

IoT & ROBÓTICA NA INDÚSTRIA 4.0

A jornada da integração entre robótica industrial e Internet das Coisas ao longo das revoluções industriais.

Indústria 1.0 — Mecanização (1784)

Introdução de máquinas a vapor e mecanização da produção. O tear mecânico revoluciona a indústria têxtil e a produção manual dá lugar às primeiras fábricas.

Indústria 2.0 — Eletrificação (1870)

Produção em massa habilitada pela energia elétrica. Linhas de montagem de Henry Ford e surgimento dos primeiros sistemas de controle elétrico nas fábricas.

Unimate — Primeiro Robô Industrial (1961)

Criado por George Devol e Joseph Engelberger, instalado na linha da General Motors. Robô de coordenadas polares que manuseava peças fundidas a 150°C — nascimento da robótica industrial.

Indústria 3.0 — Automação Digital (1969)

Introdução de CLPs, computadores industriais e sistemas SCADA. Robôs ganham controladores digitais. Surgimento do SCARA (1981), robôs Delta (1985) e sensores inteligentes.

IoT Industrial — IIoT (2010s)

Sensores conectados, protocolos MQTT e OPC UA, e plataformas de nuvem industrial (AWS IoT, Azure IoT, MindSphere) permitem que máquinas e robôs compartilhem dados em tempo real.

Indústria 4.0 — Fábricas Inteligentes (Hoje)

Convergência de robótica avançada, IoT, inteligência artificial, gêmeos digitais e edge computing. Cobots trabalham lado a lado com humanos. Produção autônoma, flexível e altamente personalizada.
